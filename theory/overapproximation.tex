\section{Preliminaries}

Fix a finite set $O$ of objects and a finite set $P$ of predicates.

\begin{definition}
An \emph{atom} is a ground expression $p(o_1,\dots,o_k)$, where $p \in P$ has arity $k \geq 0$ and $o_1,\dots,o_k \in O$.
\end{definition}

Let $F$ be the set of atoms.

\begin{definition}
A \emph{state} is a set $s \subseteq F$. The atoms in $s$ are exactly those that hold, so an atom outside $s$ is false rather than unknown.
\end{definition}

\begin{definition}
An \emph{action} $a$ is a triple $\langle \mathrm{pre}(a), \mathrm{add}(a), \mathrm{del}(a) \rangle$ of subsets of $F$. It is \emph{applicable} in a state $s$ if $\mathrm{pre}(a) \subseteq s$, and the state it leads to is then
\[
  s[a] = \bigl(s \setminus \mathrm{del}(a)\bigr) \cup \mathrm{add}(a).
\]
\end{definition}

\begin{definition}
A \emph{STRIPS planning task} over $F$ is a tuple $\Pi = \langle F,A,I,G\rangle$, where $A$ is a set of actions, $I \subseteq F$ is the initial state, and $G \subseteq F$ is the goal.
\end{definition}

Preconditions and goals are sets of atoms, so they can only require an atom to be present, never absent.

\begin{definition}
A \emph{plan} is a finite sequence $\pi = a_1 \dots a_n$ of actions. The empty plan $\varepsilon$ is applicable in every state, with $s[\varepsilon] = s$, and $\pi = a_1 \dots a_n$ with $n \geq 1$ is \emph{applicable} in $s$ if $a_1$ is applicable in $s$ and $a_2 \dots a_n$ is applicable in $s[a_1]$. The plan then leads to 
\[
  s[\pi] = s[a_1][a_2 \dots a_n].
\]
\end{definition}

\begin{definition}
A plan $\pi$ \emph{solves} $\Pi$ if it is applicable in $I$ and $G \subseteq I[\pi]$.
\end{definition}

\section{Abstraction}

Fix a set $C \subseteq O$ of objects to collapse, with $|C| > 1$, and a fresh symbol $c \notin O$.

\begin{definition}
The \emph{abstraction map} $\alpha$ is given by
\begin{align*}
  \alpha(o) &=
  \begin{cases}
    c & \text{if } o \in C,\\
    o & \text{otherwise,}
  \end{cases}
\end{align*}
and extends to atoms, states, actions, and plans by
\begin{align*}
  \alpha\bigl(p(o_1,\dots,o_k)\bigr) &= p(\alpha(o_1),\dots,\alpha(o_k)),\\
  \alpha(s) &= \{\, \alpha(f) : f \in s \,\},\\
  \alpha(a) &= \langle \alpha(\mathrm{pre}(a)), \alpha(\mathrm{add}(a)), \alpha(\mathrm{del}(a)) \rangle,\\
  \alpha(a_1 \dots a_n) &= \alpha(a_1) \dots \alpha(a_n).
\end{align*}
\end{definition}

Since $\alpha$ maps every object of $C$ onto the single symbol $c$ it is not injective, so it may identify distinct atoms, states, and actions of $\Pi$.

\begin{definition}
The \emph{abstraction} of $\Pi$ under $\alpha$ is $\Pi^\alpha = \langle F^\alpha, A^\alpha, \alpha(I), \alpha(G)\rangle$, where $F^\alpha$ is the set of atoms over $P$ and $(O \setminus C) \cup \{c\}$ and $A^\alpha = \{\, \alpha(a) : a \in A \,\}$.
\end{definition}

An atom $p(o_1,\dots,o_k) \in F$ \emph{mentions} an object $o \in O$ if $o \in \{o_1,\dots,o_k\}$.

\begin{definition}\label{def:relaxed}
The \emph{relaxation} of an action $a \in A$ is the action
\[
  a^r = \langle\, \alpha(\mathrm{pre}(a)),\ \alpha(\mathrm{add}(a)),\ \mathrm{del}^r(a) \,\rangle,
\]
where 
\[
\mathrm{del}^r(a) = \{\, \alpha(d) : d \in \mathrm{del}(a) \text{ and } d \text{ mentions no object of } C \,\}.
\]
The relaxation of a plan $\pi = a_1 \dots a_n$ is $\pi^r = a_1^r \dots a_n^r$, and the relaxation of $\Pi^\alpha$ is $\Pi^r = \langle F^\alpha, A^r, \alpha(I), \alpha(G) \rangle$, where $A^r = \{\, a^r : a \in A \,\}$.
\end{definition}

$\Pi^r$ differs from $\Pi^\alpha$ only in the delete lists of its actions, and coincides with it when no delete of $\Pi$ mentions an object of $C$.

\section{Over-approximation}

\begin{lemma}\label{lem:action}
If $a \in A$ is applicable in $s \subseteq F$, then $a^r$ is applicable in $\alpha(s)$.
\end{lemma}

\begin{proof}
By Definition~\ref{def:relaxed} the relaxation leaves preconditions untouched, so $\mathrm{pre}(a^r) = \alpha(\mathrm{pre}(a))$. From $\mathrm{pre}(a) \subseteq s$ we get $\alpha(\mathrm{pre}(a)) \subseteq \alpha(s)$, hence $\mathrm{pre}(a^r) \subseteq \alpha(s)$.
\end{proof}

\begin{lemma}\label{lem:progress}
If $a \in A$ is applicable in $s \subseteq F$, then $\alpha(s[a]) \subseteq \alpha(s)[a^r]$.
\end{lemma}

\begin{proof}
Progression gives $s[a] = (s \setminus \mathrm{del}(a)) \cup \mathrm{add}(a)$, so
\begin{align*}
  \alpha(s[a]) &= \alpha(s \setminus \mathrm{del}(a)) \cup \alpha(\mathrm{add}(a)),\\
  \alpha(s)[a^r] &= (\alpha(s) \setminus \mathrm{del}^r(a)) \cup \alpha(\mathrm{add}(a)),
\end{align*}
and it suffices to show that $\alpha(s \setminus \mathrm{del}(a)) \subseteq \alpha(s) \setminus \mathrm{del}^r(a)$. Let $f \in \alpha(s \setminus \mathrm{del}(a))$. Then $f = \alpha(g)$ for some $g \in s$ such that $g \notin \mathrm{del}(a)$, so $f \in \alpha(s)$. Suppose $f \in \mathrm{del}^r(a)$. By Definition~\ref{def:relaxed} we then have $f = \alpha(d)$ for some $d \in \mathrm{del}(a)$ mentioning no object of $C$. Since $\alpha$ fixes every object outside $C$, it fixes $d$, so $f = d$. Now $d$ is an atom over $O$ and $c \notin O$, so $f$ does not mention $c$. If $g$ mentioned an object of $C$ then $\alpha(g)$ would mention $c$, and $\alpha(g) = f$ does not. Hence $g$ mentions no object of $C$ either, so $\alpha$ fixes $g$ and $g = \alpha(g) = f = d$. This contradicts $g \notin \mathrm{del}(a)$, so $f \notin \mathrm{del}^r(a)$.
\end{proof}

\begin{lemma}\label{lem:mono}
Let $s \subseteq t$ be states and let $\pi$ be applicable in $s$. Then $\pi$ is applicable in $t$ and $s[\pi] \subseteq t[\pi]$.
\end{lemma}

\begin{proof}
By induction on the length of $\pi$. The empty plan is applicable in every state and $s[\varepsilon] = s \subseteq t = t[\varepsilon]$. For $\pi = a\pi'$, applicability of $a$ in $s$ gives $\mathrm{pre}(a) \subseteq s \subseteq t$, so $a$ is applicable in $t$, and
\[
  s[a] = (s \setminus \mathrm{del}(a)) \cup \mathrm{add}(a)
           \subseteq (t \setminus \mathrm{del}(a)) \cup \mathrm{add}(a)
            = t[a].
\]
The claim follows by applying the induction hypothesis to $\pi'$ at $s[a] \subseteq t[a]$.
\end{proof}

\begin{lemma}\label{lem:plan}
If $\pi$ is applicable in $s \subseteq F$, then $\pi^r$ is applicable in $\alpha(s)$ and $\alpha(s[\pi]) \subseteq \alpha(s)[\pi^r]$.
\end{lemma}

\begin{proof}
By induction on the length of $\pi$. For the empty plan both sides are $\alpha(s)$. Let $\pi = a\pi'$ with $a$ applicable in $s$. By Lemma~\ref{lem:action} $a^r$ is applicable in $\alpha(s)$, and by Lemma~\ref{lem:progress} $\alpha(s[a]) \subseteq \alpha(s)[a^r]$. The induction hypothesis applied to $\pi'$ at $s[a]$ gives that $\pi'^r$ is applicable in $\alpha(s[a])$ with $\alpha(s[a][\pi']) \subseteq \alpha(s[a])[\pi'^r]$. Since $\alpha(s[a]) \subseteq \alpha(s)[a^r]$, Lemma~\ref{lem:mono} lets us move both to $\alpha(s)[a^r]$, where $\pi'^r$ is applicable and
\[
  \alpha(s[\pi]) = \alpha(s[a][\pi']) \subseteq \alpha(s[a])[\pi'^r]
             \subseteq \alpha(s)[a^r][\pi'^r] = \alpha(s)[\pi^r]. \qedhere
\]
\end{proof}

\begin{theorem}
If $G \subseteq I[\pi]$, then $\alpha(G) \subseteq \alpha(I)[\pi^r]$.
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:plan} the plan $\pi^r$ is applicable in $\alpha(I)$ and $\alpha(I[\pi]) \subseteq \alpha(I)[\pi^r]$. Applying $\alpha$ to $G \subseteq I[\pi]$ gives $\alpha(G) \subseteq \alpha(I[\pi])$, so
\[
  \alpha(G) \subseteq \alpha(I[\pi]) \subseteq \alpha(I)[\pi^r]. \qedhere
\]
\end{proof}
